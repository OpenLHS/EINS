# Qu’est-ce qu’il y a dans ce fichier?

SELECT ?sujet ?prop ?objet

WHERE {
	?sujet ?prop ?objet .
}

ou

SELECT *

WHERE {
	?sujet ?prop ?objet .
}

# Les trier (ORDER BY)

SELECT ?sujet ?prop ?objet

WHERE {
	?sujet ?prop ?objet .
}
ORDER BY ?sujet


# Les trier (LIMIT)

SELECT ?sujet ?prop ?objet

WHERE {
	?sujet ?prop ?objet .
}
ORDER BY ?sujet
LIMIT 50

# Afficher toutes les données à propos de Harry Potter

SELECT ?sujet ?prop ?objet

WHERE {
	hp:Harry ?prop ?objet .
}

ou

SELECT ?sujet ?prop ?objet

WHERE {
	?sujet ?prop ?objet .
  FILTER(?sujet = hp:Harry)
}

##EXERCICE 2

# Faire la liste des élèves (hp:élève)

SELECT ?élève

WHERE {
	?élève rdf:type hp:élève .

}


# Rajouter leurs noms (foaf:name)

SELECT ?élève ?nom

WHERE {
	?élève rdf:type hp:élève .
    ?élève foaf:name ?nom

}

##EXERCICE 4

# Lister les maisons (noms) avec les noms de leurs membres
(hp:maison, rdfs:label, hp:est_membre_de, foaf:name)

SELECT ?nom_maison ?nom_membre

WHERE {
	?maison rdf:type hp:maison .
  	?membre hp:est_membre_de ?maison .
    ?maison rdfs:label ?nom_maison .
  	?membre foaf:name ?nom_membre .

}

# BONUS: Classer par nombre de membres

SELECT ?nom_maison (COUNT(?membre) AS ?nombre)

WHERE {
	?maison rdf:type hp:maison .
  	?membre hp:est_membre_de ?maison .
    ?maison rdfs:label ?nom_maison .

}
GROUP BY ?nom_maison
